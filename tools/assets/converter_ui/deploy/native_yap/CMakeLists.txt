cmake_minimum_required(VERSION 3.20)
project(ParadiseYAP LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(YAP_SOURCE /src/tools/yap CACHE PATH "YAP source checkout")
find_package(Qt6 REQUIRED COMPONENTS Core)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LIBDEFLATE_BUILD_GZIP OFF CACHE BOOL "" FORCE)
add_subdirectory(${YAP_SOURCE}/external/argparse argparse EXCLUDE_FROM_ALL)
add_subdirectory(${YAP_SOURCE}/external/GameDataStream GameDataStream EXCLUDE_FROM_ALL)
add_subdirectory(${YAP_SOURCE}/external/libdeflate libdeflate EXCLUDE_FROM_ALL)
add_subdirectory(${YAP_SOURCE}/external/yaml-cpp yaml-cpp EXCLUDE_FROM_ALL)
add_executable(YAP ${YAP_SOURCE}/src/main.cpp ${YAP_SOURCE}/src/yap.cpp
                   ${YAP_SOURCE}/src/extract.cpp ${YAP_SOURCE}/src/create.cpp)
target_include_directories(YAP PRIVATE ${YAP_SOURCE}/include)
target_compile_options(YAP PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/qt_integer_compat.h)
target_compile_options(GameDataStream PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/qt_integer_compat.h)
target_link_libraries(YAP PRIVATE argparse GameDataStream libdeflate_static yaml-cpp Qt6::Core)
