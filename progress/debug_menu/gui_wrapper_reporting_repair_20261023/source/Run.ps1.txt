function Assert-WrapperBudget([string]$Where) {

}
function Get-Sha256([string]$Path) {
    if (-not (Test-Path -LiteralPath $Path -PathType Leaf)) { throw "Pinned file is missing: $Path" }
    Assert-WrapperBudget 'before file hash'
    $hash=(Get-FileHash -LiteralPath $Path -Algorithm SHA256).Hash.ToLowerInvariant()
    Assert-WrapperBudget 'after file hash'
    return $hash
}

function Assert-RecordedFingerprints([object]$Protected) {
    foreach ($entry in $Protected.PSObject.Properties) {
        $path = [string]$entry.Name
        $expected = $entry.Value
        $exists = Test-Path -LiteralPath $path -PathType Leaf
        if ([bool]$expected.exists -ne $exists) { throw "Protected path existence changed: $path" }
        if ($exists) {
            $item = Get-Item -LiteralPath $path
            $hash = Get-Sha256 $path
            $expectedTicks = [long]([decimal]::Parse([string]$expected.mtime_ns) / 100) + 621355968000000000
            if ($item.Length -ne [long]$expected.bytes -or $hash -cne [string]$expected.sha256 -or
                $item.LastWriteTimeUtc.Ticks -ne $expectedTicks) {
                throw "Protected path changed after private staging: $path"
            }
        }
    }
}

function Get-Fingerprint([string]$Path) {
    $exists=Test-Path -LiteralPath $Path -PathType Leaf
    if(-not $exists){return [ordered]@{path=$Path;exists=$false}}
    $file=Get-Item -LiteralPath $Path
    return [ordered]@{path=$Path;exists=$true;bytes=$file.Length;sha256=(Get-Sha256 $Path);
        mtime_utc=$file.LastWriteTimeUtc.ToString('o');mtime_utc_ticks=$file.LastWriteTimeUtc.Ticks.ToString();
        mtime_ns=(([decimal]$file.LastWriteTimeUtc.Ticks-621355968000000000)*100).ToString('0',[Globalization.CultureInfo]::InvariantCulture)}
}
function Same-Fingerprint($Expected,$Actual) {
    if($null -eq $Expected -or $null -eq $Actual -or [bool]$Expected.exists -ne [bool]$Actual.exists){return $false}
    if(-not $Expected.exists){return $true}
    return [long]$Expected.bytes -eq [long]$Actual.bytes -and [string]$Expected.sha256 -ceq [string]$Actual.sha256 -and
        [string]$Expected.mtime_utc_ticks -ceq [string]$Actual.mtime_utc_ticks
}

}
function Get-NormalState($Baseline) {

    foreach($entry in $Baseline.PSObject.Properties){$state[$entry.Name]=Get-Fingerprint ([string]$entry.Name)}
    return $state
}
function Assert-NormalStateMatchesBaseline([object]$Baseline,[object]$Actual) {
    foreach($entry in $Baseline.PSObject.Properties) {
        $path=[string]$entry.Name
        if(-not $Actual.Contains($path)){throw "Normal protected state is missing: $path"}
        $expected=$entry.Value;$actualFingerprint=$Actual[$path]
        if([bool]$expected.exists -ne [bool]$actualFingerprint.exists){throw "Protected path existence changed: $path"}
        if($actualFingerprint.exists) {
            $expectedTicks=[long]([decimal]::Parse([string]$expected.mtime_ns)/100)+621355968000000000
            if([long]$expected.bytes -ne [long]$actualFingerprint.bytes -or
               [string]$expected.sha256 -cne [string]$actualFingerprint.sha256 -or
               [long]$actualFingerprint.mtime_utc_ticks -ne $expectedTicks) {
                throw "Protected path changed after private staging: $path"
            }
        }
    }
    if(@($Actual.Keys).Count -ne @($Baseline.PSObject.Properties).Count){throw 'Normal protected inventory count changed.'}
}

function Write-FallbackReceipt([string]$ReportStatus,[string]$ReporterError='') {
    $preflightErrorValue=$null
    if(-not $attempted){$preflightErrorValue=[string]$runError}
    $safeRunError=[string]$runError
    if(-not [string]::IsNullOrWhiteSpace($ReporterError)) {
        $safeRunError=$(if($safeRunError){$safeRunError+'; '}else{''})+'reporter: '+$ReporterError
    }
    if($safeRunError.Length -gt 1000){$safeRunError=$safeRunError.Substring(0,1000)}
    if($null -ne $preflightErrorValue -and $preflightErrorValue.Length -gt 1000){$preflightErrorValue=$preflightErrorValue.Substring(0,1000)}
    $safeReporterError=[string]$ReporterError
    if($safeReporterError.Length -gt 1000){$safeReporterError=$safeReporterError.Substring(0,1000)}
    $caseExitValue=-1
    if($null -ne $caseExit){$caseExitValue=[int]$caseExit}
    $caseVerdictValue=if($result){[string]$result.verdict}else{'unavailable'}
    $fallback=[ordered]@{
        schema=1;packet='gui_behavior_batch_62049';classification='INCOMPLETE_EVIDENCE_RETAINED'
        report_status=$ReportStatus;launch_attempted=[bool]$attempted;case_exit=$caseExitValue
        case_verdict=$caseVerdictValue;preflight_error=$preflightErrorValue;run_error=$safeRunError
        reporter_error=$safeReporterError
        manual_visual_review='pending';behavior_credit=$false
    }
    $compact=ConvertTo-Json -InputObject $fallback -Depth 3 -Compress
    Write-Output ('[gui-wrapper-fallback] '+$compact)
    [IO.File]::WriteAllText($receiptPath,$compact+[Environment]::NewLine,(New-Object Text.UTF8Encoding($false)))
}

            $rawManifestPath=Join-Path $runDir 'raw-private-logs.json'
            if(-not (Test-Path -LiteralPath $rawManifestPath -PathType Leaf)){throw 'run_case raw-private log manifest is missing.'}
            $rawRows=@(Get-Content -LiteralPath $rawManifestPath -Raw | ConvertFrom-Json)
            if($rawRows.Count -gt 2){throw 'Raw-private log channel count exceeds two.'}
            foreach($row in $rawRows) {
                $source=[IO.Path]::GetFullPath([string]$row.source)
                $copy=[IO.Path]::GetFullPath([string]$row.copy)
                if(-not $source.StartsWith($slotDir.TrimEnd('\')+'\',[StringComparison]::OrdinalIgnoreCase) -or
                   -not $copy.StartsWith($runDir.TrimEnd('\')+'\',[StringComparison]::OrdinalIgnoreCase)){throw 'Raw-private log manifest path escaped its owned directory.'}
                if(-not (Test-Path -LiteralPath $source -PathType Leaf) -or -not (Test-Path -LiteralPath $copy -PathType Leaf)){throw 'Raw-private log source or retained copy is missing.'}
                $sourceFingerprint=Get-Fingerprint $source
                if([long]$sourceFingerprint.bytes -ne [long]$row.bytes -or [string]$sourceFingerprint.sha256 -cne [string]$row.sha256 -or
                   (Get-Item -LiteralPath $copy).Length -ne [long]$row.bytes){throw 'Retained raw-private log does not match its source manifest.'}
                $rawLogs += [ordered]@{source=$sourceFingerprint;retained_path=$copy;retained_bytes=[long]$row.bytes;retained_sha256=[string]$row.sha256}
            }
        }
    } catch {$runError=$(if($runError){$runError+'; '}else{''})+'raw log manifest verification failed: '+$_.Exception.Message}
    try {
        $remaining=@(Get-ActiveGames); $processCheckSucceeded=$true
        if($remaining.Count -eq 0 -and $stageReceipt -and $privateBefore) {
            $backup=Join-Path $runDir 'private_slot_config.original.ini'
            if(-not (Test-Path -LiteralPath $backup -PathType Leaf)){throw 'Original private config backup is missing.'}
            Copy-Item -LiteralPath $backup -Destination $slotConfig -Force
            [IO.File]::SetLastWriteTimeUtc($slotConfig,[datetime]::new([long]$privateBefore.config.mtime_utc_ticks,[DateTimeKind]::Utc))
            $privateAfter=Get-PrivateState
            $configRestored=Same-Fingerprint $privateBefore.config $privateAfter.config
            # Only run_case owns the parked original/profile fixture lifecycle. Do not rewrite it here.
            $profileRestored=Same-Fingerprint $privateBefore.profile $privateAfter.profile
            $seedsUnchanged=(Same-Fingerprint $privateBefore.config_seed $privateAfter.config_seed) -and
                (Same-Fingerprint $privateBefore.profile_fixture $privateAfter.profile_fixture)
            if(-not $configRestored -or -not $profileRestored -or -not $seedsUnchanged){throw 'Private config/profile/seed bytes or mtimes were not restored.'}
        } elseif($attempted){throw 'A game is active or private baseline unavailable; private state/parked save retained without restoration.'}
        if($remaining.Count -eq 0 -and $attempted) {
            $ownedScriptsClean=$true
            foreach($name in @(0..23|ForEach-Object {'behavior-{0:D3}.txt' -f $_})+@('behavior-final.txt','behavior-activate.txt')) {
                $target=Join-Path $slotDir $name
                if(Test-Path -LiteralPath $target -PathType Leaf) {
                    $runCopy=Join-Path $runDir $name
                    if(-not (Test-Path -LiteralPath $runCopy -PathType Leaf) -or (Get-Sha256 $target) -cne (Get-Sha256 $runCopy)) {
                        $ownedScriptsClean=$false;$scriptCleanup += [ordered]@{path=$target;status='retained_hash_or_source_mismatch'}
                    } else {
                        $beforeRemoval=Get-Fingerprint $target
                        Remove-Item -LiteralPath $target
                        $scriptCleanup += [ordered]@{path=$target;status='removed_matching_owned_copy';before=$beforeRemoval}
                    }
                }
            }
            if(-not $ownedScriptsClean){throw 'A generated private script was changed; it was retained.'}
        }
        if($stageReceipt) {
            $normalAfter=Get-NormalState $stageReceipt.normal_protected
            Assert-NormalSlotPathsAbsent
            Assert-NormalProfileSet
            Assert-NormalStateMatchesBaseline $stageReceipt.normal_protected $normalAfter
            $normalProtectedUnchanged=$true
        }
        foreach($row in $pins.hosted_artifact.payloads) {
            $path=Join-Path $slotDir $row.name;$hash=Get-Sha256 $path
            if($hash -cne [string]$row.sha256 -or (Get-Item -LiteralPath $path).Length -ne [long]$row.bytes){throw "Private payload drift: $($row.name)"}
            $payloadsAfter += [ordered]@{name=$row.name;sha256=$hash;bytes=(Get-Item -LiteralPath $path).Length}
        }
        $cleanupVerified=$processCheckSucceeded -and $remaining.Count -eq 0 -and $terminal -and $terminal.success -eq $true -and
            $terminal.helper_cleanup_complete -eq $true -and $terminal.profile_restoration_complete -eq $true -and
            $flowStatus -and $flowStatus.owned_cleanup_complete -eq $true -and $flowStatus.input_reset_complete -eq $true
        if($terminal -and (Get-Sha256 (Join-Path $runDir 'flow/terminal-status.json')) -cne [string]$terminal.flow_status_sha256){throw 'Original flow terminal hash mismatch.'}
    } catch {$normalProtectedUnchanged=$false;$runError=$(if($runError){$runError+'; '}else{''})+$_.Exception.Message}
    # Capture actual private/normal state even if the independent checks failed.
    try {
        if(-not $privateAfter -and (Test-Path -LiteralPath $slotDir -PathType Container)){$privateAfter=Get-PrivateState}
        if($stageReceipt -and -not $normalAfter){$normalAfter=Get-NormalState $stageReceipt.normal_protected}
    } catch {$runError=$(if($runError){$runError+'; '}else{''})+'state fingerprint capture failed: '+$_.Exception.Message}
    $elapsed=((Get-Date)-$started).TotalSeconds
    $coveragePass=$result -and $result.verdict -ceq 'PASS' -and $caseExit -eq 0
    try {
        Assert-WrapperBudget 'before terminal receipt'
        $terminalReportExit=Invoke-RunReporter 'terminal' 'safety-checks-complete-before-archive'
        if($terminalReportExit -ne 0){throw 'Could not update the small terminal receipt.'}
        Assert-WrapperBudget 'before evidence archive'
        $archiveExit=Invoke-RunReporter 'archive' 'safety-checks-complete-before-archive'
        $evidenceComplete=$archiveExit -eq 0
        if(-not $evidenceComplete){
            $runError=$(if($runError){$runError+'; '}else{''})+'bounded evidence archive incomplete; see probe_receipt.json'
            if(-not (Test-Path -LiteralPath (Join-Path $runDir 'evidence-archive.json') -PathType Leaf)) {
                $reporterFailure='archive reporter did not create evidence-archive.json; inspect raw wrapper stderr'
                $runError+='; '+$reporterFailure
                try { Write-FallbackReceipt 'archive-reporter-unavailable' $reporterFailure }
                catch { Write-Output ('[gui-wrapper-fallback] '+$runError.Substring(0,[Math]::Min(1000,$runError.Length))) }
            }
        }
    } catch {
        $evidenceComplete=$false
        $reporterFailure=$_.Exception.Message
        $runError=$(if($runError){$runError+'; '}else{''})+'bounded evidence reporting failed: '+$reporterFailure
        try { Write-FallbackReceipt 'final-reporter-failed' $reporterFailure }
        catch { Write-Output ('[gui-wrapper-fallback] '+$runError.Substring(0,[Math]::Min(1000,$runError.Length))) }
    }
    $elapsed=((Get-Date)-$started).TotalSeconds
    $boundedComplete=$coveragePass -and $cleanupVerified -and $configRestored -and $profileRestored -and $seedsUnchanged -and
        $normalProtectedUnchanged -and $ownedScriptsClean -and $evidenceComplete -and -not $runError -and $elapsed -le $maxOuterSeconds
}
if(-not $boundedComplete){Write-Host "[gui-behavior-probe] incomplete evidence retained at $runDir";exit 1}
Write-Host "[gui-behavior-probe] structural case complete; actual frame interpretation remains pending: $runDir"
exit 0
